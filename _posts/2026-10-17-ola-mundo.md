(function(){
  var root=document.getElementById('it-match-jefferson');
  if(!root)return;
  var areas={
    infra:{title:'Infraestrutura e redes',keys:['redes','linux','windows','mikrotik','seguranca']},
    sistemas:{title:'Administração de sistemas',keys:['linux','windows','redes']},
    seguranca:{title:'Segurança da informação',keys:['seguranca','redes','linux']},
    suporte:{title:'Suporte técnico e serviços',keys:['suporte','windows','redes']},
    qa:{title:'QA, automação e SAP',keys:['qa','sap']}
  };
  var names={redes:'redes TCP/IP e LAN/WAN',linux:'ambientes Linux',windows:'Windows Server e Active Directory',mikrotik:'MikroTik',seguranca:'VPN e segurança da informação',suporte:'suporte técnico e indicadores de serviço',qa:'QA e automação',sap:'fundamentos SAP'};
  var studying={qa:true,sap:true};
  root.querySelector('#im-check').addEventListener('click',function(){
    var area=areas[root.querySelector('#im-area').value];
    var selected=Array.from(root.querySelectorAll('.im-chips input:checked')).map(function(el){return el.value});
    var target=selected.length?selected:area.keys;
    var established=target.filter(function(k){return !studying[k]});
    var learning=target.filter(function(k){return studying[k]});
    var exp=root.querySelector('#im-exp').value;
    var out=root.querySelector('#im-output');
    var lines=['<strong>✓ PERFIL CONSULTADO</strong>','<b>Área:</b> '+area.title];
    if(established.length)lines.push('<b>Competências profissionais:</b> '+established.map(function(k){return names[k]}).join(', ')+'.');
    if(learning.length)lines.push('<b>Em desenvolvimento:</b> '+learning.map(function(k){return names[k]}).join(', ')+'.');
    if(exp==='5')lines.push('<b>Experiência:</b> meu histórico em TI é de aproximadamente 4 anos; não afirmo ter mais de 5 anos.');
    else lines.push('<b>Experiência:</b> aproximadamente 4 anos de atuação em TI.');
    lines.push('<b>Contexto:</b> 2º Tenente, oficial temporário da FAB, com atuação em gestão de infraestrutura e serviços de TI no ILA.');
    out.innerHTML=lines.join('<br><br>');
    out.hidden=false;
  });
})();
</script>
